server {
    server_name dcbsas.com www.dcbsas.com 2.25.71.62 _;

    root /var/www/html;
    index index.html;

    # Proxy reverso para el backend - DEBE IR ANTES de location /
    location /api/ {
        proxy_pass http://localhost:5000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 300s;
    }

    # Proxy para evidencias (archivos estáticos del backend)
    location /evidencias/ {
        proxy_pass http://localhost:5000/evidencias/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # SPA fallback
    location / {
        try_files $uri $uri/ /index.html;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/dcbsas.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/dcbsas.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}
server {
    if ($host = www.dcbsas.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = dcbsas.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    listen [::]:80;
    server_name dcbsas.com www.dcbsas.com 2.25.71.62 _;
    return 404; # managed by Certbot




}
